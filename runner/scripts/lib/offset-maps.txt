# offset-maps.txt — the one table of how every committed target.h offset is
# derived from a kernel build. Read by both paths, so they cannot drift:
#
#   runner/scripts/harvest-live.sh   live, from a rooted device
#   tools/pixel-image/derive_offsets.py   offline, from a public boot.img
#
# Both feed runner/scripts/lib/offset_rules.py, which implements one resolver
# per rule. A row is `MACRO <rule fields...>`; `#` comments and blank lines are
# ignored. Sections are named by the env var offset_report.py reads.
#
# Anything expressible here belongs here. An addend or an anchor rule written
# only in a header comment is a derivation nobody can re-run, which is how
# SLIDE_TRACE_MARK_IP_OFF — the single constant every root run consumes —
# went unchecked by offsets.report for 19 targets.

# MACRO  enumerator
#
# Values that are enumerators in the kernel rather than offsets. They are

#
# sizeof(struct mm_struct) is 0x3c0 but the cache strides 0x400, and

[PTRMAP]
# MACRO  deref    base-macro    byte-offset
# MACRO  findptr  target-macro  back-offset
#
# Offsets no symbol table exposes. `ashmem_compat_ioctl` and `ashmem_misc` are
# static, so kallsyms drops them — on a rooted device too — and BTF has no
# data-symbol addresses; both are reachable by chasing a pointer through
# ashmem_fops in the Image. `findptr` demands exactly one hit, so an ambiguous
# match is an UNRESOLVED rather than a wrong answer.
#
# The byte-offset field may name another macro instead of a literal, and here
# it must: file_operations.compat_ioctl is 0x58 on android14-6.1 and 0x50 on
# android15-6.6, so the literal that used to sit here read the wrong slot on
# every 6.6 target. No live capture had covered one, so nothing caught it.
ASHMEM_COMPAT_IOCTL_OFF deref ASHMEM_FOPS_OFF FOPS_COMPAT_IOCTL_OFF
ASHMEM_MISC_FOPS_OFF findptr ASHMEM_FOPS_OFF 0


[CODEMAP]
# MACRO  cbz-to-warn  symbol  span     -> offset WITHIN symbol
# MACRO  self-addr    symbol  0        -> offset within the IMAGE
#
# Offsets that only the instruction stream carries. The two rules report
# different things because the two macros hold different things: a kprobe
# oracle is placed relative to its function, while SLIDE_TRACE_MARK_IP_OFF is
# an image offset the payload subtracts from a leaked pointer.
#
# self-addr finds the adrp/add pair inside <symbol> whose computed address
# lands back inside <symbol> — how the compiler materialises _THIS_IP_. That
# is the value tracing_mark_write stores into print_entry.ip, which is the
# whole KASLR leak (cves/lib/kaslr/README.md). It is not KMI-stable — 0x164 on
# android14-6.1, 0x15c on android15-6.6 — which is exactly why it has to be a
# rule and cannot be a SYMMAP addend.
POSIX_CPU_TIMER_DEL_RACE_OFF cbz-to-warn posix_cpu_timer_del 4
SLIDE_TRACE_MARK_IP_OFF self-addr tracing_mark_write 0

[STACKMAP]
# MACRO  stack-overlay  <first_word>   -> a SIGNED word index
#
# Not an address: a relation between two kernel stack frames. The pselect
# reclaim vehicle overlays fd_set bitmaps onto the freed rt_mutex_waiter, and
# this is the word of core_sys_select()'s stack_fds the waiter lands on, minus
# the index the payload's word map starts at ([pselect_map].first_word in
# runner/stages/entry.ghostlock@6.1/stage.toml).
#
# Per build, not per kernel version. The landing word is decided by the
# compiler's frame layout and by whether the intermediate call survived
# inlining, so two builds of one kernel version can differ and one of them can
# be unusable; inheriting the value from a sibling flavour has no basis. The
# rule reports UNRESOLVED when a chain link was inlined away, which is the same
# answer as "this build cannot use the pselect vehicle".
PSELECT_WAITER_WORD_SHIFT stack-overlay 2
