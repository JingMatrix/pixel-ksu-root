# offset-maps.txt — the one table of how every committed target.h offset is
# derived from a kernel build. Read by both paths, so they cannot drift:
#
#   runner/scripts/harvest-live.sh   live, from a rooted device
#   tools/pixel-image/derive_offsets.py   offline, from a public boot.img
#
# Both feed runner/scripts/lib/offset_rules.py, which implements one resolver
# per rule. A row is `MACRO <rule fields...>`; `#` comments and blank lines are
# ignored. Sections are named by the env var offset_report.py reads.
#
# Anything expressible here belongs here. An addend or an anchor rule written
# only in a header comment is a derivation nobody can re-run and offsets.report
# cannot check, no matter how central the constant it derives -- SLIDE_TRACE_
# MARK_IP_OFF, the single constant every root run consumes, is exactly such a
# case.

[PTRMAP]
# MACRO  deref    base-macro    byte-offset
# MACRO  findptr  target-macro  back-offset
#
# Offsets no symbol table exposes. `ashmem_compat_ioctl` is static, so kallsyms
# drops it, and BTF has no data-symbol addresses; it is reached by chasing a
# pointer through ashmem_fops in the Image. `findptr` demands exactly one hit,
# so an ambiguous match is an UNRESOLVED rather than a wrong answer.
#
# findptr's "exactly one hit" guard is not safe for ASHMEM_MISC_FOPS_OFF when
# the miscdevice.fops field is a RELATIVE relocation whose addend is not
# stored inline (qgki-5.4): the scan can lock onto a different, coincidental
# inline pointer to ashmem_fops and return it as "unique". ashmem_misc IS a
# kallsyms data symbol on 5.4, so it is derived in SYMMAP as
# `ashmem_misc + miscdevice.fops(0x10)` instead. findptr remains the right
# rule only for a build that both strips ashmem_misc and stores the pointer
# inline.
#
# The byte-offset field may name another macro instead of a literal, and here
# it must: file_operations.compat_ioctl is 0x58 on android14-6.1 and 0x50 on
# android15-6.6, so a literal offset here would read the wrong slot on one of
# the two families; naming the macro keeps the derivation correct on both.
ASHMEM_COMPAT_IOCTL_OFF deref ASHMEM_FOPS_OFF FOPS_COMPAT_IOCTL_OFF


[CODEMAP]
# MACRO  cbz-to-warn  symbol  span     -> offset WITHIN symbol
# MACRO  self-addr    symbol  0        -> offset within the IMAGE
#
# Offsets that only the instruction stream carries. The two rules report
# different things because the two macros hold different things: a kprobe
# oracle is placed relative to its function, while SLIDE_TRACE_MARK_IP_OFF is
# an image offset the payload subtracts from a leaked pointer.
#
# self-addr finds the adrp/add pair inside <symbol> whose computed address
# lands back inside <symbol> — how the compiler materialises _THIS_IP_. That
# is the value tracing_mark_write stores into print_entry.ip, which is the
# whole KASLR leak (cves/lib/kaslr/README.md). It is not KMI-stable — 0x164 on
# android14-6.1, 0x15c on android15-6.6 — which is exactly why it has to be a
# rule and cannot be a SYMMAP addend.
POSIX_CPU_TIMER_DEL_RACE_OFF cbz-to-warn posix_cpu_timer_del 4
SLIDE_TRACE_MARK_IP_OFF self-addr tracing_mark_write 0

[STACKMAP]
# MACRO  stack-overlay  <first_word>   -> a SIGNED word index
#
# Not an address: a relation between two kernel stack frames. The pselect
# reclaim vehicle overlays fd_set bitmaps onto the freed rt_mutex_waiter, and
# this is the word of core_sys_select()'s stack_fds the waiter lands on, minus
# the index the payload's word map starts at ([pselect_map].first_word in
# runner/stages/entry.ghostlock@6.1/stage.toml).
#
# Per build, not per kernel version. The landing word is decided by the
# compiler's frame layout and by whether the intermediate call survived
# inlining, so two builds of one kernel version can differ and one of them can
# be unusable; inheriting the value from a sibling flavour has no basis. The
# rule reports UNRESOLVED when a chain link was inlined away, which is the same
# answer as "this build cannot use the pselect vehicle".
PSELECT_WAITER_WORD_SHIFT stack-overlay 2

# MACRO  stack-bytes-sigreturn  (no argument)   -> a SIGNED byte offset
#
# The sigreturn vehicle's analogue: raising a signal on this thread copies a
# flat struct user_fpsimd_state.vregs onto __arm64_sys_rt_sigreturn's own
# stack frame (or, on a build where restore_fpsimd_context survives as its
# own function, that function's frame instead -- offset_rules.py's
# _sigreturn_delta() picks between the two shapes by which symbols this
# build's kallsyms actually has), and this is the byte distance from that
# copy's start to where the freed rt_mutex_waiter lands -- no word index, no
# first_word, since the copy is a flat buffer rather than pselect's
# word-indexed fd_set. A target's own runtime gate
# (SIGRETURN_WAITER_OFF_GATE_FAIL, cve-2026-43499-ghostlock/main.c) refuses
# whatever this value turns out to be if it does not fit inside
# [0, SIGRETURN_VREGS_BYTES - FAKE_WAITER_LAYOUT_SIZE] -- this rule only
# measures the fact, never judges whether it is usable.
SIGRETURN_WAITER_OFF stack-bytes-sigreturn

# MACRO  stack-bytes-clone3  (no argument)   -> a SIGNED byte offset
#
# The clone3 vehicle's analogue: __arm64_sys_clone3(2) copies the fixed-size
# `struct clone_args` uapi argument onto copy_clone_args_from_user()'s own
# stack frame (via the inlined copy_struct_from_user()), and this is the
# byte distance from that copy's start to where the freed rt_mutex_waiter
# lands -- no word index, flat buffer, same shape as SIGRETURN_WAITER_OFF
# above. offset_rules.py's _CLONE3_WITNESSES comment explains the one thing
# that differs by build here: whether the copy calls __arch_copy_from_user
# directly or an out-of-line per-TU "_copy_from_user" clone -- resolving the
# latter needs load_duplicate_addrs()/load_duplicate_addrs_from_text() run
# first (both derive_offsets.py and harvest-live.sh's offset_report.py do
# this unconditionally, so no extra wiring is needed by a caller of either).
#
# struct clone_args is 0x58 bytes on a build with set_tid/set_tid_size/
# cgroup (flags@0, pidfd@8, child_tid@0x10, parent_tid@0x18,
# exit_signal@0x20, stack@0x28, stack_size@0x30, tls@0x38, set_tid@0x40,
# set_tid_size@0x48, cgroup@0x50) and 0x40 bytes without them (same first 8
# fields, CLONE_ARGS_SIZE_VER0); copy_clone_args_from_user checks
# args.exit_signal immediately after this copy and returns -EINVAL before
# clone3_args_valid()/_do_fork() are ever reached if that check fails --
# whether a vehicle built on this offset reaches that same check, a
# different (later, weaker) one, or corrupts a field no check reads at all,
# depends on where WAITER_TASK_OFF/WAITER_LOCK_OFF (STRUCTMAP, above) plus
# this delta land inside those field boundaries; this rule measures the byte
# distance only, never which field that lands on.
CLONE3_WAITER_OFF stack-bytes-clone3

# MACRO  stack-bytes-process-vm  (no argument)   -> a SIGNED byte offset

# The process_vm_readv/process_vm_writev vehicle's analogue: process_vm_rw()
# copies its "remote" struct iovec[UIO_FASTIOV] argument onto its own stack
# frame -- a FULL flat buffer, not a narrow 2-field target like clone3, same
# shape as SIGRETURN_WAITER_OFF above. Unlike clone3 this never needs
# load_duplicate_addrs(): both kernel shapes checked (android14-6.1's
# iovec_from_user, qgki-5.4's older rw_copy_check_uvector) take the fast
# array's address as a direct argument, offset_rules.py's
# _PROCESS_VM_WITNESSES comment has the details. A target's own runtime gate
# (PROCESS_VM_WAITER_OFF_GATE_FAIL, cve-2026-43499-ghostlock/main.c) refuses

# [0, PROCESS_VM_IOVSTACK_BYTES - FAKE_WAITER_LAYOUT_SIZE].
PROCESS_VM_WAITER_OFF stack-bytes-process-vm

# MACRO  stack-bytes-bpf  (no argument)   -> a SIGNED byte offset
#
# The bpf(2) vehicle's analogue: every bpf(2) command shares one front-door
# copy, __sys_bpf()'s/__do_sys_bpf()'s own stack-local `union bpf_attr attr`.

# fully owns, not an ABI-fixed struct like clone3's, so it needs a bounded

# clone3's exact match. Picks between two chain shapes by which survives in
# kallsyms (offset_rules.py's _bpf_delta()): android14-6.1/android15-6.6

# __do_sys_bpf, with __sys_bpf itself inlined away. A target's own runtime
# gate (bpf_gate_check(), cve-2026-43499-ghostlock/fops_bpf.c) refuses
# whatever this value turns out to be if either written field would not fit
# inside [0, BPF_ATTR_BYTES - 8].
BPF_WAITER_OFF stack-bytes-bpf

# MACRO  stack-bytes-mcast  (no argument)   -> a SIGNED byte offset
#
# The IPv4 MCAST_JOIN_SOURCE_GROUP vehicle: one setsockopt(2) copies a
# 264-byte struct group_source_req onto do_ip_setsockopt()'s own stack frame

# copy, unlike the narrow two-field vehicles. offset_rules.py's _mcast_delta()

# it lands in the complete-bridge window only where the setsockopt buffer
# overlaps the deeper futex waiter (a 5.15-GKI property; on 6.1/5.4 it lands

MCAST_WAITER_OFF stack-bytes-mcast

#

# genuine tables hold -- so one row covers both kinds of build and a target
